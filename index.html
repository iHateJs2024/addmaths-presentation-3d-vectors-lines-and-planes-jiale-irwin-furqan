from manim import * from manim_slides import Slide class TestScene(Slide): def construct(self): # Slide 1: Intro Text t = Text("Hello World!") self.play(Write(t, run_time=2)) self.next_slide() # Slide 2: Draw Circle and Dot c = Circle(radius=3) self.play(Create(c)) d = Dot() d.move_to(c.get_start()) self.play(Write(d)) # Slide 3: Seamlessly Looping Path self.next_slide(loop=True) self.play(MoveAlongPath(d, c)) self.next_slide() # Slide 4: Clear Screen & Outro self.play(*[FadeOut(m) for m in self.mobjects]) t2 = Text("Bye!") self.play(Write(t2, run_time=1))
